import pandas as pd

data = [
    ["Chicago","Q1","Mobile",605],
    ["Chicago","Q1","Modem",825],
    ["Toronto","Q1","Mobile",395],
    ["Vancouver","Q2","Mobile",450],
]

df = pd.DataFrame(data, columns=["Location","Time","Item","Amount"])

# Roll-up (sum by Location)
rollup = df.groupby("Location")["Amount"].sum()

print("Roll-up Output:\n")
print(rollup)

import pandas as pd

data = [
    ["Chicago","Q1","Mobile",605],
    ["Toronto","Q1","Mobile",395],
    ["Vancouver","Q2","Modem",450],
    ["Toronto","Q2","Modem",200],
]

df = pd.DataFrame(data, columns=["Location","Time","Item","Amount"])

# Dice (multiple conditions)
dice_df = df[
    (df["Location"].isin(["Toronto","Vancouver"])) &
    (df["Time"].isin(["Q1","Q2"]))
]

print("Dice Output:\n")
print(dice_df)
